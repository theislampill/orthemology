\newif\iforthemologyTaskThirteenEmptyBibliography
\def\orthemologyTaskThirteenZero{0}
\def\orthemologyTaskThirteenPatchLayout{%
  \let\orthemologyTaskThirteenOriginalTwoColumn\twocolumn
  \def\orthemologyTaskThirteenArgumentlessTwoColumn{%
    \if@twocolumn
    \else
      \orthemologyTaskThirteenOriginalTwoColumn
    \fi
  }%
  \renewcommand{\twocolumn}{%
    \@ifnextchar[%
      {\orthemologyTaskThirteenOriginalTwoColumn}%
      {\orthemologyTaskThirteenArgumentlessTwoColumn}%
  }%
  \let\orthemologyTaskThirteenOriginalTheBibliography\thebibliography
  \let\orthemologyTaskThirteenOriginalEndTheBibliography\endthebibliography
  \renewenvironment{thebibliography}[1]{%
    \def\orthemologyTaskThirteenBibliographyCount{##1}%
    \ifx\orthemologyTaskThirteenBibliographyCount\orthemologyTaskThirteenZero
      \orthemologyTaskThirteenEmptyBibliographytrue
    \else
      \orthemologyTaskThirteenEmptyBibliographyfalse
      \orthemologyTaskThirteenOriginalTheBibliography{##1}%
    \fi
  }{%
    \iforthemologyTaskThirteenEmptyBibliography
    \else
      \orthemologyTaskThirteenOriginalEndTheBibliography
    \fi
  }%
}
\AtBeginDocument{%
  \setlength{\tabcolsep}{3pt}%
  \orthemologyTaskThirteenPatchLayout
}
\makeatother
